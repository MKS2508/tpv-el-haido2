### M5 — Tenant resolution middleware (~20min)

`apps/server/src/middleware/tenant.ts`:

```ts
import { Elysia } from 'elysia'
import { db } from '../db/client'
import { projects } from '@mks2508/release-hub-shared'
import { eq } from 'drizzle-orm'
import type { ITenantContext } from '@mks2508/release-hub-shared'
import { tenantLogger } from '../lib/logger'

/** In-memory cache for project lookup (TTL 60s). Reduces DB hits per request. */
const projectCache = new Map<string, { project: ITenantContext; expiresAt: number }>()
const CACHE_TTL_MS = 60_000

/**
 * Tenant resolution middleware.
 *
 * Reads Host header → extracts subdomain → looks up project by slug → attaches to context.
 *
 * - Subdomain `admin` → bypass (admin routes don't need tenant context)
 * - Subdomain `localhost` / IP → bypass (local dev)
 * - Any other subdomain → SELECT projects WHERE slug = subdomain
 * - Project not found → 404 (handled by route, this middleware just attaches null)
 */
export function createTenantMiddleware() {
  return new Elysia({ name: 'tenant-middleware' })
    .derive({ as: 'global' }, async ({ request }) => {
      const host = request.headers.get('host') ?? ''
      const subdomain = extractSubdomain(host)

      if (!subdomain || subdomain === 'admin' || subdomain === 'localhost') {
        return { tenant: null as ITenantContext | null }
      }

      const cached = projectCache.get(subdomain)
      if (cached && cached.expiresAt > Date.now()) {
        return { tenant: cached.project }
      }

      const [row] = await db
        .select({ id: projects.id, slug: projects.slug, name: projects.name })
        .from(projects)
        .where(eq(projects.slug, subdomain))
        .limit(1)

      if (!row) {
        tenantLogger.warn('Project not found for subdomain', { subdomain })
        return { tenant: null as ITenantContext | null }
      }

      const tenant: ITenantContext = {
        projectId: row.id,
        projectSlug: row.slug,
        projectName: row.name,
      }
      projectCache.set(subdomain, { project: tenant, expiresAt: Date.now() + CACHE_TTL_MS })
      return { tenant }
    })
}

/**
 * Extracts subdomain from `<sub>.releases.mks2508.systems` host.
 * Returns null if host doesn't match expected pattern.
 */
function extractSubdomain(host: string): string | null {
  const cleaned = host.split(':')[0]  // strip port
  if (cleaned === 'localhost' || /^\d+\.\d+\.\d+\.\d+$/.test(cleaned)) return null
  const parts = cleaned.split('.')
  if (parts.length < 3) return null  // not a subdomain.domain.tld
  return parts[0]
}
```

```ts
import { writeFile, mkdir, stat } from 'node:fs/promises'
import { join } from 'node:path'
import { tryCatchAsync, type Result } from '@mks2508/no-throw'
import { HubErrorCode } from '../lib/error-codes'
import { storageLogger } from '../lib/logger'

/**
 * Binary storage interface. Allows swapping local FS for S3-compatible (R2/B2) in future.
 */
export interface IBinaryStorage {
  write(key: string, content: Buffer | Uint8Array): Promise<Result<{ url: string }, { code: string; message: string }>>
  read(key: string): Promise<Result<Buffer, { code: string; message: string }>>
  delete(key: string): Promise<Result<void, { code: string; message: string }>>
}

/** Local filesystem storage. Suitable for single-node deploys with bind mount. */
export class LocalFsStorage implements IBinaryStorage {
  constructor(private baseDir: string, private publicUrlPrefix: string) {}

  async write(key: string, content: Buffer | Uint8Array) {
    return tryCatchAsync(async () => {
      const fullPath = join(this.baseDir, key)
      const dir = fullPath.substring(0, fullPath.lastIndexOf('/'))
      await mkdir(dir, { recursive: true })
      await writeFile(fullPath, content)
      storageLogger.info('Wrote artifact', { key, size: content.byteLength })
      return { url: `${this.publicUrlPrefix}/${key}` }
    }, HubErrorCode.StorageWriteFailed)
  }

  async read(key: string) {
    const { readFile } = await import('node:fs/promises')
    return tryCatchAsync(() => readFile(join(this.baseDir, key)), HubErrorCode.StorageReadFailed)
  }

  async delete(key: string) {
    const { unlink } = await import('node:fs/promises')
    return tryCatchAsync(() => unlink(join(this.baseDir, key)).then(() => undefined), HubErrorCode.StorageWriteFailed)
  }
}
```

```ts
    arch: string,
    currentVersion: string
  ): Promise<Result<IUpdateResponse | null, { code: HubErrorCodeT; message: string }>> {
    if (!SUPPORTED_TARGETS.includes(target as any) || !SUPPORTED_ARCHES.includes(arch as any)) {
      return fail(HubErrorCode.UnsupportedTarget, `Unsupported: ${target}/${arch}`)
    }

    return tryCatchAsync(async () => {
      const [latest] = await db
        .select()
        .from(releases)
        .where(and(eq(releases.projectId, projectId), eq(releases.target, target), eq(releases.arch, arch)))
        .orderBy(desc(releases.pubDate))
        .limit(1)

      if (!latest) {
        updateLogger.info('No releases found', { projectId, target, arch })
        return null
      }

      if (!semver.valid(currentVersion) || !semver.valid(latest.version)) {
        updateLogger.warn('Invalid semver', { currentVersion, latestVersion: latest.version })
        return null
      }

      const hasUpdate = semver.gt(latest.version, currentVersion)
      updateLogger.info('Update check', { projectId, target, arch, currentVersion, latestVersion: latest.version, hasUpdate })
      if (!hasUpdate) return null

      return {
        version: latest.version,
        notes: latest.notes ?? '',
        pub_date: latest.pubDate.toISOString(),
        url: latest.url,
        signature: latest.signature,
      }
    }, HubErrorCode.DatabaseError)
  }
}
```

```ts
export interface IReleaseHubClientOptions {
  baseUrl: string  // e.g. https://haido.releases.mks2508.systems
  fetch?: typeof fetch
}

export class ReleaseHubClient {
  constructor(private opts: IReleaseHubClientOptions) {}

  async checkUpdate(target: string, arch: string, currentVersion: string): Promise<IUpdateResponse | null> {
    const f = this.opts.fetch ?? fetch
    const res = await f(`${this.opts.baseUrl}/api/updates/${target}/${arch}/${currentVersion}`)
    if (res.status === 204) return null
    if (!res.ok) throw new Error(`Update check failed: ${res.status}`)
    return res.json()
  }
```
