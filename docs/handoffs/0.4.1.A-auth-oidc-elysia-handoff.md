```ts
import type { JWTPayloadSpec } from '@elysiajs/jwt'

export interface IAuthConfig {
  /** Pocket ID issuer URL, e.g. https://auth-provider.mks2508.systems */
  issuerUrl: string
  /** OIDC client_id registrado en Pocket ID */
  clientId: string
  /** OIDC client_secret. Optional para public clients (PKCE only). */
  clientSecret?: string
  /** Redirect URI registrado en Pocket ID, e.g. https://admin.mydomain.com/auth/callback/oidc */
  redirectUri: string
  /** Secret para firmar session JWT (HS256). 32+ bytes random. */
  sessionSecret: string
  /** Subs (user IDs Pocket ID) permitidos como admins. Si vacío → cualquier user autenticado pasa. */
  adminSubs?: string[]
  /** URL base del admin SPA para redirigir tras callback. Si no, redirige a / */
  adminUiUrl?: string
  /** Cookie domain para session JWT, e.g. .mydomain.com (cross-subdomain). Default: undefined (host-only). */
  cookieDomain?: string
  /** Bypass total de auth (útil para dev local). Default: false */
  bypass?: boolean
  /** Session JWT TTL en segundos. Default: 604800 (7 días) */
  sessionTtl?: number
}

export interface IAuthContext {
  isAuthenticated: boolean
  sub?: string
  email?: string
  name?: string
  picture?: string
  /** Original Pocket ID access_token (para llamar a APIs del provider si hace falta) */
  accessToken?: string
}

export interface IAuthStatusResponse {
  authenticated: boolean
  sub: string | null
  email: string | null
  name: string | null
  picture: string | null
}

export interface ISessionPayload extends JWTPayloadSpec {
  access_token: string
  refresh_token?: string
  sub: string
  email?: string
  name?: string
  picture?: string
}
```

#### M3.5 — `src/middleware.ts`

`createAuthMiddleware(config)` — devuelve un Elysia que:

1. Si `config.bypass === true` → derive `{isAuthenticated: true, sub: 'dev', ...}`.
2. Lee cookie `oidc_session` (httpOnly).
3. Intenta `session.verify(cookie.oidc_session.value)` → si verde, deriva `IAuthContext` desde el payload.
4. Si cookie falla o no existe, lee `Authorization: Bearer ...`. Intenta:
   - Primero como session JWT (HS256, mismo secret) → derive context.
   - Si no, como access_token de Pocket ID → `verifyAccessToken(token, config)` → derive context. Si `config.adminSubs` existe y `sub` no está → return `{isAuthenticated: false}`.
5. Default: `{isAuthenticated: false}`.

```ts
export function requireAuth() {
  return ({ isAuthenticated, set }: IAuthContext & { set: { status: number } }) => {
    if (!isAuthenticated) {
      set.status = 401
      throw new Error('UNAUTHORIZED')
    }
  }
}
```

#### M3.6 — `src/routes.ts`

Plugin con prefix configurable (default `/auth`), endpoints:

| Endpoint | Método | Descripción |
|---|---|---|
| `/auth/health` | GET | Returns `{ ok: true, issuer }` |
| `/auth/login/oidc` | GET | Genera state + verifier + challenge (S256), persiste en cookie `oidc_state` (httpOnly, 10min), 302 a Pocket ID. Acepta `?token_mode=true&return_to=URL` para CLI loopback (devuelve fragment con token en lugar de cookie). |
| `/auth/callback/oidc` | GET | Verifica state, exchange code → tokens, firma session JWT, set cookie + 302 a `adminUiUrl` (o fragment si token_mode). |
| `/auth/logout` | POST | Borra cookie. Opcional: 302 a `end_session_endpoint` del discovery. |
| `/auth/status` | GET | Lee cookie/Bearer, returns `IAuthStatusResponse`. |

#### M3.7 — `src/plugin.ts` (factory)

```ts
import { Elysia } from 'elysia'
import { createSessionJwtPlugin } from './jwt'
import { createAuthMiddleware } from './middleware'
import { createAuthRoutes } from './routes'
import type { IAuthConfig } from './types'

export function createAuthPlugin(config: IAuthConfig, opts?: { routePrefix?: string }) {
  const prefix = opts?.routePrefix ?? '/auth'
  return new Elysia({ name: 'mks-auth-oidc' })
    .use(createSessionJwtPlugin(config))
    .use(createAuthMiddleware(config))
    .group(prefix, (app) => app.use(createAuthRoutes(config)))
}
```

```ts
// tests/plugin.test.ts
import { describe, it, expect } from 'vitest'
import { Elysia } from 'elysia'
import { createAuthPlugin, requireAuth } from '../src'

describe('createAuthPlugin', () => {
  it('mounts /auth/health and returns issuer', async () => {
    const app = new Elysia().use(createAuthPlugin({
      issuerUrl: 'https://example.com',
      clientId: 'test',
      redirectUri: 'http://localhost/cb',
      sessionSecret: 'a'.repeat(32),
    }))
    const res = await app.handle(new Request('http://localhost/auth/health'))
    const body = await res.json()
    expect(body.ok).toBe(true)
    expect(body.issuer).toBe('https://example.com')
  })

  it('requireAuth returns 401 without auth', async () => {
    const app = new Elysia()
      .use(createAuthPlugin({
        issuerUrl: 'https://example.com',
        clientId: 'test',
        redirectUri: 'http://localhost/cb',
        sessionSecret: 'a'.repeat(32),
      }))
      .get('/protected', ({ set }) => 'ok', { beforeHandle: requireAuth() })
    const res = await app.handle(new Request('http://localhost/protected'))
    expect(res.status).toBe(401)
  })

  it('bypass mode allows access', async () => {
    const app = new Elysia()
      .use(createAuthPlugin({
        issuerUrl: 'https://example.com',
        clientId: 'test',
        redirectUri: 'http://localhost/cb',
        sessionSecret: 'a'.repeat(32),
        bypass: true,
      }))
      .get('/protected', () => 'ok', { beforeHandle: requireAuth() })
    const res = await app.handle(new Request('http://localhost/protected'))
    expect(res.status).toBe(200)
  })
})
```

```ts
// tests/jwt.test.ts
import { describe, it, expect } from 'vitest'
import { Elysia } from 'elysia'
import { createSessionJwtPlugin } from '../src/jwt'

describe('session jwt', () => {
  it('signs and verifies a payload roundtrip', async () => {
    const app = new Elysia()
      .use(createSessionJwtPlugin({
        issuerUrl: '', clientId: '', redirectUri: '',
        sessionSecret: 'b'.repeat(32),
      }))
      .get('/sign', async ({ session }) => session.sign({ sub: 'user-1', access_token: 'xyz' }))
      .get('/verify', async ({ session, query }) => session.verify(query.t as string))
    const signRes = await app.handle(new Request('http://localhost/sign'))
    const token = await signRes.text()
    expect(token).toBeTruthy()
    const verifyRes = await app.handle(new Request(`http://localhost/verify?t=${token}`))
    const payload = await verifyRes.json()
    expect(payload.sub).toBe('user-1')
  })
})
```
